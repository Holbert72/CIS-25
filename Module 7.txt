#include <iostream>
#include <sstream>
#include <vector>
#include <string>

using namespace std;

vector<string> splitCSVLine(const string &line) {
    vector<string> result;
    stringstream ss(line);
    string cell;

    while (getline(ss, cell, ',')) {
        result.push_back(cell);
    }
    return result;
}

int main() {
    string csvData =
        "Name,UIC,Latitude,Longitude,Country\n"
        "Château-Arnoux – St-Auban,8775123,44.061565,5.997373,FR\n"
        "York,7082630,53.957966,-1.093159,GB\n"
        "Exeter Central,7057550,50.726471,-3.532973,GB\n"
        "Chevremont,8718425,47.621629,6.921473,FR\n"
        "Branná,5433112,50.147004,17.014842,CZ\n"
        "Březnice,,5476934,49.558057,13.963859,CZ\n"
        "Tyseley,7045180,52.453819,-1.839568,GB\n"
        "Rhymney,7040270,51.759064,-3.289494,GB\n"
        "Menton,8775648,43.774444,7.4933,FR\n"
        "Calamocha Nueva,7167021,40.903354,-1.30028,ES";

    istringstream file(csvData);
    string line;

    while (getline(file, line)) {
        vector<string>fields=splitCSVLine(line);

        for (const auto &field : fields) {
            cout<<field<<" | ";
        }
        cout<<"\n";
    }

    return 0;
}
