#include <iostream>
#include <fstream>
#include <string>

using namespace std;

class Item {
public:
    string itemName;
    float itemPrice;
    int itemQuantity;
}; 
    
void showMenu() {
    cout<<"===== SHOPPING CART MENU ====="<<endl;
    cout<<"Type what number option you want. Like 1, 2, or 3\n";
    cout<<"1. Add item to cart\n2. View checkout\n3. Exit\n";
    cout<<"Enter your choice: ";
}

int main() {
    string itemName;
    float itemPrice;
    int itemQuantity;
    int menuOption;
    
    fstream file("cart.txt");
    
    do {
        showMenu();
        cin>>menuOption;
        if (menuOption==1) {
	    int continueAdding;
            do {
                cout<<"Enter item name: ";
                cin>>itemName;
                cout<<"Enter item price: ";
                cin>>itemPrice;
                cout<<"Enter item quantity: ";
                cin>>itemQuantity;
                ofstream file("cart.txt", ios::app);
    		if (file.is_open()) {
        	    file<<itemName<<" "<<itemPrice<<" "<<itemQuantity<<"\n";
        	    file.close();
        	    cout<<"Item saved to cart\n";
    		} else {
        	    cout<<"Unable to open file for writing."<<endl;
    		}
                cout<<"Type 1 if you would like to continue adding items, or 0 to return to the menu: ";
                cin>>continueAdding;
            } while (continueAdding==1);
        }
        if (menuOption==2) {
            fstream file("cart.txt");
	    float total=0;
    	    cout<<"\n===== CHECKOUT ====="<<endl;
  	    while (file>>itemName>>itemPrice>>itemQuantity) {
                float itemTotal=itemPrice*itemQuantity;
        	total+=itemTotal;
        	cout<<itemName<<" - $"<<itemPrice<<" x "<<itemQuantity<<" = $"<<itemTotal<<endl;
    	    }
    	    cout<<"----------------------------------------"<<endl;
    	    cout<<"Total: $"<<total<<endl;
    	    file.close();
        }
        if (menuOption==3) {
	        cout << "Thank you for shopping!\n";
            return 0;
        }
    } while (menuOption!=3);
    return 0;
}
