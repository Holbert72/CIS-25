#include <iostream>

using namespace std;

int main() {
    
    int age;
    
    cout<<"What is your age?\n";
    cin>>age;
    
    if (age>=18) {
        
        cout<<"You are an adult.";
        
    } else if (age>=13) {
        
        cout<<"You are a teen.";

    } else if (age>=5) {
        
        cout<<"You are a kid.";
        
    } else if (age>=1) {
        
        cout<<"You are a toddler.";
        
    } else {
        
        cout<<"You are just born.";
        
    }
    
}

Output 1: 

What is your age?
42
You are an adult. 

Output 2:

What is your age?

Output 3:

What is your age?
10
You are a kid.

Output 4:

What is your age?
0
You are just born.
