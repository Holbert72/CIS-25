using namespace std;

class Song {
public:
    int id;
    string name;
    
    Song() : id(0), name("") {}
    
    Song(int sI, string sN) {
        id = sI;
        name = sN;
    }
    
    ~Song() {}
    
    void printSong() {
        cout<<"ID: "<<id<<", Name: "<<name<<endl;
    }
};

class Playlist {
public:
    int id;
    string name;
    
    Playlist() : id(0), name("") {}
    
    Playlist(int pI, string pN) {
        id = pI;
        name = pN;
    }
    
    ~Playlist() {}
    
    void printPlaylist() {
        cout<<"ID: "<<id<<", Name: "<<name<<endl;
    }
};

int binarySearch(Song songs[], int size, int searchID) {
    int left = 0;
    int right = size - 1;
    
    while (left <= right) {
        int mid = left + (right - left) / 2;
        
        if (songs[mid].id == searchID) {
            return mid;
        }
        
        if (songs[mid].id < searchID) {
            left = mid + 1;
        } else {
            right = mid - 1;
        }
    }
    
    return -1;
}

        capacity *= 2;
